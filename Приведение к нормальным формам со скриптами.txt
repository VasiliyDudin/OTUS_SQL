3) Приведем таблицу ко второй нормальной форме (2НФ). Для этого создадим три взаимосвязанные таблицы Orders, Clients, Products.

CREATE TABLE public."Clients"
(
    "Client_ID" INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    "Client_Name" text NOT NULL,
    "Client_Phone" text,
    "Address" text NOT NULL
);

ALTER TABLE IF EXISTS public."Clients"
    OWNER to postgres;


CREATE TABLE public."Products"
(
    "Code_Product" INTEGER PRIMARY KEY,
    "Name_Product" text NOT NULL,
    "Price_Product" numeric NOT NULL
);

ALTER TABLE IF EXISTS public."Products"
    OWNER to postgres;



CREATE TABLE public."Orders"
(
    "Order_Number" INTEGER NOT NULL,
    "Count"        INTEGER,
    "Price_Order"  numeric,
    "Order_Date"   DATE NOT NULL,
    "Client_ID"    INTEGER NOT NULL,
    "Product_ID"   INTEGER NOT NULL,

	PRIMARY KEY ("Order_Number", "Product_ID"),
    
    -- Внешний ключ на Clients(Client_ID)
    CONSTRAINT fk_orders_client
        FOREIGN KEY ("Client_ID")
        REFERENCES public."Clients" ("Client_ID")
        ON DELETE RESTRICT
        ON UPDATE CASCADE,
    
    -- Внешний ключ на Products(Code_Product)
    CONSTRAINT fk_orders_product
        FOREIGN KEY ("Product_ID")
        REFERENCES public."Products" ("Code_Product")
        ON DELETE RESTRICT
        ON UPDATE CASCADE
);

INSERT INTO public."Orders" (
    "Order_Number",
    "Order_Date",
    "Count",
    "Price_Order",
    "Client_ID",
    "Product_ID"
)
SELECT 
    o."Order_Number",
    o."Order_Date",
    o."Count"::INTEGER,
    o."Price_Order",
    c."Client_ID",
    o."Code_Product"::INTEGER
FROM public."OrdersOld" o
JOIN public."Clients" c 
    ON c."Client_Name" = o."Client_Name" 
    AND c."Client_Phone" = o."Client_Phone"
    AND c."Address" = o."Address"
ORDER BY o."Order_Number", o."Code_Product";

3) Приведем таблицы к третьей нормальной форме (3НФ). В целом, после выше проведённых изменений, БД уже сейчас находится в 3НФ. Все транзитивные зависимости устранены, но для более наглядной демонстрации вынесем Order_Date из таблицы Orders в отдельную таблицу.

CREATE TABLE public."Dates" (
    "Date_ID"        INTEGER PRIMARY KEY,
    "Full_Date"      DATE NOT NULL UNIQUE,
    "Year"           INTEGER NOT NULL,
    "Quarter"        INTEGER NOT NULL,      -- 1, 2, 3, 4
    "Month"          INTEGER NOT NULL,      -- 1-12
    "Month_Name"     TEXT NOT NULL,         -- Январь, Февраль, ...
    "Week"           INTEGER NOT NULL,      -- Номер недели в году (1-53)
    "Day"            INTEGER NOT NULL,      -- День месяца (1-31)
    "Day_Of_Week"    INTEGER NOT NULL,      -- 1-7 (Пн=1, Вс=7)
    "Day_Name"       TEXT NOT NULL,         -- Понедельник, Вторник, ...
    "Is_Weekend"     BOOLEAN NOT NULL,      -- TRUE для субботы/воскресенья
    "Is_Holiday"     BOOLEAN DEFAULT FALSE, -- TRUE для праздников
    "Year_Month"     INTEGER NOT NULL,      -- ГГГГММ (для группировок)
    "Year_Quarter"   TEXT NOT NULL          -- '2024-Q1'
);

-- Индексы для производительности
CREATE INDEX idx_dates_year ON public."Dates" ("Year");
CREATE INDEX idx_dates_month ON public."Dates" ("Month");
CREATE INDEX idx_dates_quarter ON public."Dates" ("Quarter");
CREATE INDEX idx_dates_full_date ON public."Dates" ("Full_Date");
