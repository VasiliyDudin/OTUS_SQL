CREATE OR REPLACE PROCEDURE insert_clients(cnt INT, batch_size INT DEFAULT 1000)
LANGUAGE plpgsql
AS $$
DECLARE
	first_name TEXT;
    last_name TEXT;
    email TEXT;
    phone TEXT;
    start_cnt INT := 0;
BEGIN
    WHILE start_cnt < cnt LOOP
		first_name := 'Клиент ' || start_cnt;
        last_name := 'Фамилия ' || start_cnt;
        email := CONCAT(MD5(RANDOM()::TEXT), '@yandex.ru');
        phone := LPAD(FLOOR(RANDOM() * 10000000000)::TEXT, 10, '0');
		
        INSERT INTO public."Clients" ("First name", "Last name", "E-mail", "Phone", "Address")
        VALUES (first_name, last_name, email, phone, NULL);

        start_cnt := start_cnt + 1;

        IF start_cnt % batch_size = 0 THEN
            COMMIT;
        END IF;
    END LOOP;

    COMMIT;
END $$;

Вызов - CALL insert_clients(1000, 100);
