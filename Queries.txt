########################## Task DZ ###############################################

1) SELECT c."Name", jsonb_path_query(c.additional_info, '$.languages') as languages FROM public."Cinema" as c;
2) SELECT c."Name", jsonb_path_query(c.additional_info, '$.budget ? (@ > 100000000)') as languages FROM public."Cinema" as c
3) SELECT c."Id", jsonb_build_object('full_name', c."First name" || ' ' || c."Last name",
								'contact', jsonb_build_object( 'E-mail', c."E-mail", 'Phone', c."Phone")) as JSON_Obj 
   FROM public."Clients" as c

4) UPDATE public."Clients"
   SET preferences = jsonb_set(
    			preferences,
    			'{preferred_genres}',
    			COALESCE(preferences->'preferred_genres', '[]'::jsonb) || '"Drama"'::jsonb
   )
   WHERE preferences->>'newsletter' = 'true';

5) SELECT c."Genre",
    	ROUND(AVG((c.additional_info->>'budget')::NUMERIC), 2) AS avg_budget,
    	COUNT(c.*) AS films_count
   FROM public."Cinema" as c
   GROUP BY c."Genre"
   ORDER BY avg_budget DESC;
