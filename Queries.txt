1) create or replace function GetMovieDurationInHours(p_cinema_id  int)
returns numeric(10, 2)
as $$
declare
	v_duration numeric := 0.0;
	v_name_cinema text;
begin

	select ROUND((p."Duration"/60.0)::numeric, 2), p."Name" into v_duration, v_name_cinema
	from public."Cinema" as p where "Id" = p_cinema_id;

	raise notice 'Продолжительность фильма - % составляет - % часов.', v_name_cinema, v_duration;

	return v_duration;
end;
$$ language plpgsql;


2) create or replace function GetMoviesByDirector(p_director_name text)
 returns table(Name_cinema text, Release_Year int, Genre text)
as $$
declare
begin
 	return query
    select 
        p."Name",
        EXTRACT(YEAR FROM p."Release Year")::INT,
        p."Genre"
    from public."Cinema" as p
    where p.additional_info->>'director' = p_director_name;
	
end;
$$ language plpgsql;
